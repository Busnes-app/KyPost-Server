// Reverse tabnabbing through a mail body while KyDrive is paired (COOP same-origin-allow-popups).
// A disposable KyPost server paired by environment to a stand-in KyDrive (https://drive.example.test,
// never contacted), real sign-in, the kypost-demo-IMAP fixture. /api/mail/body for the opened message is
// replaced with a hostile body; the server, frontend, CSP, COOP and the sandboxed mail frame are real.
// Each link is clicked for real and the opened page reports whether window.opener is non-null.
// Usage: node opener-driver.mjs <binary> <frontend dist> <fixture dir> <expect: open|closed>
import { mkdir, readFile, readdir, rm, writeFile } from 'node:fs/promises';
import { existsSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { secret, tempDir, pw, waitReady, startServer, settle } from '/home/yoshi/git/ky-redesign/captures/drivers/lib.mjs';

const engines = await pw('KyCalendar/web');
const ENGINE = process.env.ENGINE || 'chromium';
const PORT = '5891', IMAP = '5893', SMTP = '5894', HTTPS = '5895';
const ORIGIN = `http://localhost:${PORT}`;
const EVIL = 'https://evil.example.test';
const [binary, frontend, fixtureDir] = process.argv.slice(2, 5).map((p) => resolve(p ?? ''));
const EXPECT = process.argv[5];
if (!['open', 'closed'].includes(EXPECT)) throw new Error('usage: opener-driver.mjs <binary> <dist> <fixture> <open|closed>');

const PNG = 'data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mP8z8BQDwAEhQGAhKmMIQAAAABJRU5ErkJggg==';
const CASES = {
  'area rel=opener': `<p><img src="${PNG}" usemap="#m" width="240" height="40"><map name="m"><area shape="rect" coords="0,0,240,40" href="${EVIL}/area" target="_blank" rel="opener"></map></p>`,
  'area (no rel)': `<p><img src="${PNG}" usemap="#m2" width="240" height="40"><map name="m2"><area shape="rect" coords="0,0,240,40" href="${EVIL}/area2"></map></p>`,
  'a rel=opener': `<p><a href="${EVIL}/a" target="_blank" rel="opener">Open the invoice</a></p>`,
  'svg a href rel=opener': `<p><svg width="240" height="40"><a href="${EVIL}/svg" target="_blank" rel="opener"><rect width="240" height="40" fill="#888"></rect></a></svg></p>`,
  'svg a xlink:href': `<p><svg width="240" height="40" xmlns:xlink="http://www.w3.org/1999/xlink"><a xlink:href="${EVIL}/xlink" target="_blank"><rect width="240" height="40" fill="#888"></rect></a></svg></p>`,
  'svg a xlink:href rel=opener': `<p><svg width="240" height="40" xmlns:xlink="http://www.w3.org/1999/xlink"><a xlink:href="${EVIL}/xlink2" target="_blank" rel="opener"><rect width="240" height="40" fill="#888"></rect></a></svg></p>`,
  'area target=named': `<p><img src="${PNG}" usemap="#m3" width="240" height="40"><map name="m3"><area shape="rect" coords="0,0,240,40" href="${EVIL}/area3" target="w1"></map></p>`,
  'svg a xlink:href target=named': `<p><svg width="240" height="40" xmlns:xlink="http://www.w3.org/1999/xlink"><a xlink:href="${EVIL}/xlink3" target="w2"><rect width="240" height="40" fill="#888"></rect></a></svg></p>`,
};
// id is stripped by the sanitizer, so targets are structural.
const TARGETS = { 'area rel=opener': 'img[usemap="#m"]', 'area (no rel)': 'img[usemap="#m2"]', 'a rel=opener': 'p > a', 'svg a href rel=opener': 'svg >> nth=0', 'svg a xlink:href': 'svg >> nth=1', 'svg a xlink:href rel=opener': 'svg >> nth=2', 'area target=named': 'img[usemap="#m3"]', 'svg a xlink:href target=named': 'svg >> nth=3' };
const BODY = `<html><body><p>Hostile message.</p>${Object.values(CASES).join('')}</body></html>`;

const dir = await tempDir('post-opener');
const d = (...p) => join(dir, ...p);
for (const p of ['config', 'state', 'logs', 'secrets', 'keys', 'public']) await mkdir(d(p), { recursive: true });
await writeFile(d('secrets', 'kydrive'), secret() + secret());
const USER = 'charlie';
const pws = { admin: secret(), user: secret() };
let fixture, srv, failed = false;
const results = [];
try {
  fixture = startServer(process.execPath, ['src/index.js'], {
    cwd: fixtureDir,
    env: { PATH: process.env.PATH, BIND_ADDRESS: '127.0.0.1', IMAP_PORT: IMAP, SMTP_PORT: SMTP, HTTPS_PORT: HTTPS,
      ALLOWED_CLIENT_IPS: '127.0.0.1,::1', TLS_KEY_DIR: d('keys'), TLS_PUBLISH_DIR: d('public') },
  });
  for (let i = 0; i < 80 && !(await readdir(d('public'))).length; i++) await new Promise((r) => setTimeout(r, 250));
  srv = startServer(binary, [], {
    cwd: dir,
    env: { PATH: process.env.PATH, HOME: dir, CONFIG_DIR: d('config'), STATE_DIR: d('state'), LOG_DIR: d('logs'), SECRET_DIR: d('secrets'),
      FRONTEND_DIR: frontend, WEB_PORT: PORT, SSL_CERT_DIR: d('public'), SANDBOX_PRIVATE_HOSTS: 'localhost', KYPOST_DISABLE_AI_SORTING: 'true',
      KYPOST_KYDRIVE_URL: 'https://drive.example.test', KYPOST_KYDRIVE_SECRET_FILE: d('secrets', 'kydrive'),
      PUSH_RELAY_URL: '', APNS_RELAY_URL: '' },
  });
  await waitReady(`${ORIGIN}/api/health`, 90000);
  const firstRun = d('config', 'first-run-password.txt');
  for (let i = 0; i < 40 && !existsSync(firstRun); i++) await new Promise((r) => setTimeout(r, 250));
  const bootstrap = (await readFile(firstRun, 'utf8')).match(/^password: (.+)$/m)[1];
  await rm(firstRun);

  const browser = await engines[ENGINE].launch();
  const ctx = await browser.newContext({ viewport: { width: 1440, height: 900 } });
  const page = await ctx.newPage();
  globalThis.__page = page;
  const login = async (user, password) => {
    await page.goto(`${ORIGIN}/login`);
    await page.getByLabel('Username', { exact: true }).fill(user);
    await page.getByLabel('Password', { exact: true }).fill(password);
    await page.getByRole('button', { name: 'Sign in', exact: true }).click();
  };
  const changePassword = async (next) => {
    await page.getByLabel('New password', { exact: true }).fill(next);
    await page.locator('button.auth-submit').click();
    await page.waitForURL('**/read', { timeout: 60000 });
  };
  await login('admin', bootstrap);
  await changePassword(pws.admin);
  await page.goto(`${ORIGIN}/admin/server`);
  await page.getByRole('tab', { name: 'Manage Users' }).click();
  await page.getByLabel('Your account password').fill(pws.admin);
  const create = page.locator('form.users-create-card');
  await create.getByLabel('Username').fill(USER);
  const temp = secret();
  await create.getByLabel('Temporary Password').fill(temp);
  await create.getByRole('button', { name: 'Create User' }).click();
  await page.locator('.users-list-card').getByText(USER, { exact: true }).first().waitFor();
  await ctx.clearCookies();
  await login(USER, temp);
  await changePassword(pws.user);
  await page.goto(`${ORIGIN}/settings/mail`);
  const card = page.locator('.config-card').first();
  const field = (label) => card.locator('label', { has: page.locator(`div:text-is("${label}")`) }).locator('input');
  await field('Host').fill('localhost');
  await field('Port').fill(IMAP);
  await field('Username').fill('charlie@kypost-demo.local');
  await field('Password or App Password').fill(secret());
  await field('Mailbox').fill('INBOX');
  await field('SMTP Host (optional)').fill('localhost');
  await field('SMTP Port (optional)').fill(SMTP);
  await card.getByRole('button', { name: 'Save', exact: true }).click();
  await card.getByText('IMAP configuration saved.').waitFor({ timeout: 30000 });

  // The opened page, served for any path on the stand-in origin.
  await ctx.route(`${EVIL}/**`, (r) => r.fulfill({ status: 200, contentType: 'text/html', body: '<!doctype html><title>evil</title>' }));
  await ctx.route('**/api/mail/body?*', async (r) => {
    const res = await r.fetch();
    return r.fulfill({ response: res, json: { ...(await res.json()), body: BODY, bodyMode: 'html' } });
  });

  const doc = await page.goto(`${ORIGIN}/read`);
  const coop = (await doc.allHeaders())['cross-origin-opener-policy'];
  console.log(`COOP on /read: ${coop}`);
  if (coop !== 'same-origin-allow-popups') throw new Error('server is not paired: COOP is ' + coop);
  await page.locator('.inbox-unit', { hasText: 'Quarterly deck attached' }).first().click({ timeout: 90000 });
  await page.locator('iframe[sandbox]').first().waitFor({ timeout: 60000 });
  await page.getByRole('button', { name: 'More actions' }).click();
  await page.getByRole('button', { name: 'Show Remote Content' }).click();
  await page.keyboard.press('Escape');
  const frameEl = page.locator('iframe[sandbox]').first();
  console.log(`frame sandbox: ${await frameEl.getAttribute('sandbox')}`);
  const frame = page.frameLocator('iframe[sandbox]').first();
  await frame.locator('p > a').first().waitFor({ timeout: 30000 });
  await settle(page, 500);
  // What survived sanitizing, for the record.
  console.log('rendered body:', await frame.locator('body').evaluate((b) => b.innerHTML.replace(/data:image[^"]+/g, 'data:…')));

  for (const [name, sel] of Object.entries(TARGETS)) {
    const target = frame.locator(sel);
    if (!(await target.count())) { results.push({ name, result: 'removed by sanitizer' }); continue; }
    const popupP = ctx.waitForEvent('page', { timeout: 5000 }).catch(() => null);
    await target.click();
    const popup = await popupP;
    if (!popup) { results.push({ name, result: 'no new window' }); continue; }
    await popup.waitForLoadState('domcontentloaded');
    const opener = await popup.evaluate(() => window.opener !== null);
    // With an opener, try the tabnabbing itself: navigate the KyPost tab from the opened page.
    let tabnabbed = false;
    if (opener) {
      const err = await popup.evaluate((u) => { try { window.opener.top.location = u; return 'no error'; } catch (e) { return String(e); } }, `${EVIL}/phish`);
      const reach = await popup.evaluate(() => { try { return { frameOrigin: window.opener.location.origin, sameOriginWithTop: !!window.opener.top.document }; } catch (e) { return String(e); } });
      console.log(name, 'top navigation:', err, JSON.stringify(reach));
      const ferr = await popup.evaluate((u) => { try { window.opener.location = u; return 'no error'; } catch (e) { return String(e); } }, `${EVIL}/framed`);
      await page.waitForTimeout(1500);
      const frameUrls = page.frames().map((f) => f.url());
      console.log(name, 'frame navigation:', ferr, JSON.stringify(frameUrls));
      if (frameUrls.includes(`${EVIL}/framed`)) tabnabbed = 'mail frame replaced';
      await page.waitForURL(`${EVIL}/phish`, { timeout: 5000 }).then(() => { tabnabbed = true; }, () => {});
    }
    results.push({ name, url: popup.url(), opener, tabnabbed });
    if (tabnabbed) { await page.goBack(); await frame.locator('p > a').first().waitFor({ timeout: 30000 }).catch(() => {}); }
    await popup.close();
  }
  await browser.close();
} catch (e) { await globalThis.__page?.screenshot({ path: '/tmp/w2-fix-area/fail.png' }).catch(() => {}); throw e; } finally {
  await srv?.stop();
  await fixture?.stop();
  await rm(dir, { recursive: true, force: true });
}
for (const r of results) console.log(JSON.stringify(r));
const leaked = results.filter((r) => r.opener);
if (EXPECT === 'closed' && leaked.length) { console.log(`FAIL: ${leaked.length} window(s) hold window.opener`); failed = true; }
if (EXPECT === 'open' && !results.find((r) => r.name === 'area rel=opener')?.opener) { console.log('NOT REPRODUCED: area rel=opener has no opener'); failed = true; }
if (!results.length) failed = true;
console.log(failed ? 'RESULT: FAIL' : 'RESULT: PASS');
process.exit(failed ? 1 : 0);
